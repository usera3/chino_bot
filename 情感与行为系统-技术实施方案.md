```python
# new-bot/models/emotion_models.py

from sqlalchemy import Column, Integer, String, Float, DateTime, Text, JSON
from sqlalchemy.ext.declarative import declarative_base
from datetime import datetime
import json

Base = declarative_base()

class EmotionState(Base):
    """情感状态表 - 记录用户的情感和行为状态"""
    __tablename__ = 'emotion_states'
    
    id = Column(Integer, primary_key=True, autoincrement=True)
    user_id = Column(String(50), nullable=False, unique=True, index=True)
    
    # === PAD三维情感 ===
    pleasure = Column(Float, default=0.3)      # 愉悦度 (-1.0 ~ 1.0)
    arousal = Column(Float, default=0.0)       # 唤醒度 (-1.0 ~ 1.0)
    dominance = Column(Float, default=0.0)     # 支配度 (-1.0 ~ 1.0)
    
    # === 生理状态 ===
    energy = Column(Float, default=80.0)       # 体力 (0 ~ 100)
    hunger = Column(Float, default=30.0)       # 饥饿感 (0 ~ 100)
    sleepiness = Column(Float, default=20.0)   # 困倦度 (0 ~ 100)
    
    # === 社交状态 ===
    loneliness = Column(Float, default=40.0)   # 孤独感 (0 ~ 100)
    intimacy_level = Column(Float, default=0.5) # 亲密度 (0.0 ~ 1.0)
    
    # === 认知状态 ===
    stress = Column(Float, default=20.0)       # 压力 (0 ~ 100)
    interest = Column(Float, default=50.0)     # 兴趣度 (0 ~ 100)
    
    # === 派生状态 ===
    mood_label = Column(String(20), default='平静')  # 当前心情标签
    current_activity = Column(String(50), nullable=True)  # 当前活动
    current_activity_start = Column(DateTime, nullable=True)  # 活动开始时间
    
    # === 时间戳 ===
    last_update = Column(DateTime, default=datetime.now, onupdate=datetime.now)
    last_chat_time = Column(DateTime, nullable=True)
    created_at = Column(DateTime, default=datetime.now)
    
    def to_dict(self):
        """转换为字典"""
        return {
            "user_id": self.user_id,
            "pad": {
                "pleasure": self.pleasure,
                "arousal": self.arousal,
                "dominance": self.dominance
            },
            "physical": {
                "energy": self.energy,
                "hunger": self.hunger,
                "sleepiness": self.sleepiness
            },
            "social": {
                "loneliness": self.loneliness,
                "intimacy_level": self.intimacy_level
            },
            "cognitive": {
                "stress": self.stress,
                "interest": self.interest
            },
            "mood_label": self.mood_label,
            "current_activity": self.current_activity,
            "last_update": self.last_update.isoformat() if self.last_update else None
        }
```

```python
# new-bot/services/emotion_service.py

from typing import Dict, List, Optional, Tuple
from datetime import datetime, time, timedelta
import random
from nonebot.log import logger
from models.emotion_models import EmotionState, DailyEvent, ActivityLog, get_db_session

class EmotionService:
    """情感与行为服务"""
    
    def __init__(self):
        """初始化情感服务"""
        self.personality = {
            "openness": 0.6,
            "conscientiousness": 0.8,
            "extraversion": 0.3,
            "agreeableness": 0.7,
            "neuroticism": 0.4
        }
        
        # 状态缓存（避免频繁数据库访问）
        self._state_cache: Dict[str, Tuple[EmotionState, datetime]] = {}
        self._cache_ttl = 60  # 缓存60秒
        
        logger.success("✅ 情感服务初始化完成")
    
    # ==================== 核心API ====================
    
    async def get_emotion_state(self, user_id: str) -> EmotionState:
        """
        获取用户的情感状态
        优先从缓存读取，缓存过期则从数据库读取
        """
        # 检查缓存
        if user_id in self._state_cache:
            cached_state, cache_time = self._state_cache[user_id]
            if (datetime.now() - cache_time).total_seconds() < self._cache_ttl:
                return cached_state
        
        # 从数据库获取
        session = get_db_session()
        try:
            state = session.query(EmotionState).filter_by(user_id=user_id).first()
            
            if not state:
                # 首次创建默认状态
                state = self._create_default_state(user_id)
                session.add(state)
                session.commit()
                session.refresh(state)
            
            # 更新缓存
            self._state_cache[user_id] = (state, datetime.now())
            
            return state
        finally:
            session.close()
    
    async def update_emotion_state(
        self,
        user_id: str,
        updates: Dict[str, float]
    ) -> EmotionState:
        """
        更新情感状态
        
        Args:
            user_id: 用户ID
            updates: 要更新的字段和增量值 {"pleasure": +0.2, "energy": -10, ...}
        
        Returns:
            更新后的情感状态
        """
        session = get_db_session()
        try:
            state = session.query(EmotionState).filter_by(user_id=user_id).first()
            
            if not state:
                state = self._create_default_state(user_id)
                session.add(state)
            
            # 应用更新
            for field, delta in updates.items():
                if hasattr(state, field):
                    current_value = getattr(state, field)
                    new_value = current_value + delta
                    
                    # 限制范围
                    if field in ['pleasure', 'arousal', 'dominance']:
                        new_value = max(-1.0, min(1.0, new_value))
                    elif field in ['energy', 'hunger', 'sleepiness', 'loneliness', 'stress', 'interest']:
                        new_value = max(0, min(100, new_value))
                    elif field == 'intimacy_level':
                        new_value = max(0.0, min(1.0, new_value))
                    
                    setattr(state, field, new_value)
            
            # 更新派生状态
            state.mood_label = self._derive_mood_label(state)
            state.last_update = datetime.now()
            
            session.commit()
            session.refresh(state)
            
            # 更新缓存
            self._state_cache[user_id] = (state, datetime.now())
            
            logger.info(f"[Emotion] 更新情感状态: {user_id} | {updates}")
            
            return state
        finally:
            session.close()
    
    async def get_current_activity(self, user_id: str) -> Optional[str]:
        """获取当前活动"""
        state = await self.get_emotion_state(user_id)
        
        # 检查活动是否需要更新
        if state.current_activity and state.current_activity_start:
            # 如果活动时间超过2小时，自动结束
            if (datetime.now() - state.current_activity_start).total_seconds() > 7200:
                await self._end_current_activity(user_id)
                return None
        
        return state.current_activity
    
    async def start_activity(
        self,
        user_id: str,
        activity_name: str,
        activity_type: str
    ):
        """开始一个新活动"""
        # 结束当前活动
        await self._end_current_activity(user_id)
        
        # 更新状态
        session = get_db_session()
        try:
            state = session.query(EmotionState).filter_by(user_id=user_id).first()
            if state:
                state.current_activity = activity_name
                state.current_activity_start = datetime.now()
                session.commit()
                
                # 记录活动日志
                log = ActivityLog(
                    user_id=user_id,
                    activity_name=activity_name,
                    activity_type=activity_type,
                    start_time=datetime.now()
                )
                session.add(log)
                session.commit()
                
                logger.info(f"[Activity] {user_id} 开始活动: {activity_name}")
        finally:
            session.close()
    
    async def trigger_random_event(self, user_id: str) -> Optional[Dict]:
        """
        触发随机事件
        
        Returns:
            事件信息字典，如果没有触发则返回None
        """
        # 30%概率触发事件
        if random.random() > 0.3:
            return None
        
        # 获取当前情感状态（影响事件类型概率）
        state = await self.get_emotion_state(user_id)
        
        # 根据当前情感调整事件概率
        if state.pleasure > 0.3:
            # 心情好时，更容易遇到正面事件
            event_weights = [0.5, 0.3, 0.2]  # [positive, neutral, negative]
        elif state.pleasure < -0.3:
            # 心情差时，更容易遇到负面事件
            event_weights = [0.2, 0.3, 0.5]
        else:
            event_weights = [0.35, 0.35, 0.3]
        
        event_type = random.choices(
            ['positive', 'neutral', 'negative'],
            weights=event_weights
        )[0]
        
        # 选择事件
        event = self._get_random_event(event_type)
        
        if event:
            # 保存事件
            await self._save_event(user_id, event_type, event)
            
            # 应用情感影响
            if 'impact' in event:
                await self.update_emotion_state(user_id, event['impact'])
            
            logger.info(f"[Event] {user_id} 触发事件: {event['desc']}")
            
            return event
        
        return None
    
    async def apply_circadian_rhythm(self, user_id: str):
        """
        应用昼夜节律影响
        根据当前时间自动调整生理状态
        """
        current_hour = datetime.now().hour
        state = await self.get_emotion_state(user_id)
        
        updates = {}
        
        # 根据时间段调整状态
        if 6 <= current_hour < 9:
            # 早晨：逐渐清醒
            updates['arousal'] = 0.05
            updates['sleepiness'] = -2
        elif 9 <= current_hour < 12:
            # 上午：精力充沛
            updates['arousal'] = 0.03
            updates['energy'] = -1
        elif 12 <= current_hour < 14:
            # 午后：困倦
            updates['sleepiness'] = 3
            updates['arousal'] = -0.05
        elif 14 <= current_hour < 18:
            # 下午：恢复
            updates['sleepiness'] = -1
            updates['energy'] = -2
        elif 18 <= current_hour < 21:
            # 傍晚：逐渐疲惫
            updates['energy'] = -2
            updates['arousal'] = -0.02
        elif 21 <= current_hour < 23:
            # 晚上：准备休息
            updates['sleepiness'] = 2
            updates['arousal'] = -0.05
        else:
            # 深夜：应该睡觉
            updates['sleepiness'] = 5
            updates['energy'] = -3
        
        # 饥饿感随时间增加
        last_update = state.last_update
        hours_since_update = (datetime.now() - last_update).total_seconds() / 3600
        updates['hunger'] = hours_since_update * 5  # 每小时+5饥饿感
        
        # 孤独感随时间增加
        if state.last_chat_time:
            hours_since_chat = (datetime.now() - state.last_chat_time).total_seconds() / 3600
            if hours_since_chat > 3:
                updates['loneliness'] = (hours_since_chat - 3) * 2  # 3小时后开始增加
        
        await self.update_emotion_state(user_id, updates)
    
    async def get_emotion_context(self, user_id: str) -> str:
        """
        生成情感上下文文本（供AI使用）
        
        Returns:
            格式化的情感状态描述
        """
        state = await self.get_emotion_state(user_id)
        
        # 获取最近的事件
        recent_events = await self._get_recent_events(user_id, hours=2)
        
        # 构建上下文
        context_parts = []
        
        # 1. 当前时间和活动
        current_time = datetime.now().strftime('%H:%M')
        context_parts.append(f"当前时间: {current_time}")
        
        if state.current_activity:
            context_parts.append(f"正在: {state.current_activity}")
        
        # 2. 情感状态描述
        mood_desc = self._describe_emotion(state)
        context_parts.append(f"心情: {mood_desc}")
        
        # 3. 生理状态
        physical_desc = []
        if state.energy < 30:
            physical_desc.append("有点累")
        elif state.energy > 80:
            physical_desc.append("精力充沛")
        
        if state.sleepiness > 60:
            physical_desc.append("困")
        
        if state.hunger > 70:
            physical_desc.append("有点饿")
        
        if physical_desc:
            context_parts.append(f"状态: {', '.join(physical_desc)}")
        
        # 4. 最近事件
        if recent_events:
            event_desc = recent_events[-1]['event_desc']
            context_parts.append(f"最近: {event_desc}")
        
        # 5. 社交状态
        if state.loneliness > 60:
            context_parts.append("有点寂寞")
        
        return " | ".join(context_parts)
    
    # ==================== 内部方法 ====================
    
    def _create_default_state(self, user_id: str) -> EmotionState:
        """创建默认情感状态"""
        return EmotionState(
            user_id=user_id,
            pleasure=0.3,
            arousal=0.0,
            dominance=0.0,
            energy=80.0,
            hunger=30.0,
            sleepiness=20.0,
            loneliness=40.0,
            intimacy_level=0.5,
            stress=20.0,
            interest=50.0,
            mood_label='平静'
        )
    
    def _derive_mood_label(self, state: EmotionState) -> str:
        """从PAD值派生心情标签"""
        p, a, d = state.pleasure, state.arousal, state.dominance
        
        if p > 0.5 and a > 0.3:
            return "开心"
        elif p > 0.3 and a < 0:
            return "平静"
        elif p < -0.3 and a < -0.3:
            return "疲惫"
        elif p < -0.3 and a > 0.3:
            return "焦虑"
        elif a < -0.5:
            return "困倦"
        elif p > 0.5 and a > 0.6:
            return "兴奋"
        else:
            return "一般"
    
    def _describe_emotion(self, state: EmotionState) -> str:
        """生成情感描述"""
        p, a = state.pleasure, state.arousal
        
        if p > 0.5 and a > 0.3:
            return "心情不错，有点兴奋"
        elif p > 0.3 and abs(a) < 0.3:
            return "挺平静的"
        elif p < -0.3 and a < 0:
            return "有点低落，还有点累"
        elif a < -0.5:
            return "困..."
        elif p < 0 and a > 0.3:
            return "有点烦躁"
        else:
            return "还好"
    
    def _get_random_event(self, event_type: str) -> Optional[Dict]:
        """获取随机事件"""
        events = {
            'positive': [
                {"desc": "看到窗外有只可爱的小猫", "impact": {"pleasure": 0.3}},
                {"desc": "听到喜欢的音乐", "impact": {"pleasure": 0.2, "arousal": 0.1}},
                {"desc": "完成了一件拖了很久的事", "impact": {"dominance": 0.2, "stress": -15}},
                {"desc": "天气突然变好了", "impact": {"pleasure": 0.15}},
                {"desc": "找到了之前丢的东西", "impact": {"pleasure": 0.25, "stress": -10}},
            ],
            'neutral': [
                {"desc": "听到街上有声音", "impact": {"arousal": 0.05}},
                {"desc": "想起以前的事情", "impact": {"pleasure": 0.0}},
                {"desc": "时间过得真快", "impact": {}},
            ],
            'negative': [
                {"desc": "东西找不到了", "impact": {"pleasure": -0.2, "stress": 10}},
                {"desc": "突然感觉有点累", "impact": {"arousal": -0.3, "energy": -10}},
                {"desc": "想到有点烦心的事", "impact": {"pleasure": -0.15, "stress": 5}},
            ]
        }
        
        event_list = events.get(event_type, [])
        return random.choice(event_list) if event_list else None
    
    async def _save_event(self, user_id: str, event_type: str, event: Dict):
        """保存事件到数据库"""
        session = get_db_session()
        try:
            db_event = DailyEvent(
                user_id=user_id,
                event_type=event_type,
                event_desc=event['desc'],
                emotion_impact=event.get('impact', {})
            )
            session.add(db_event)
            session.commit()
        finally:
            session.close()
    
    async def _get_recent_events(self, user_id: str, hours: int = 2) -> List[Dict]:
        """获取最近的事件"""
        session = get_db_session()
        try:
            from sqlalchemy import desc
            cutoff_time = datetime.now() - timedelta(hours=hours)
            
            events = session.query(DailyEvent).filter(
                DailyEvent.user_id == user_id,
                DailyEvent.event_date >= cutoff_time
            ).order_by(desc(DailyEvent.event_date)).limit(5).all()
            
            return [
                {
                    "event_desc": e.event_desc,
                    "event_type": e.event_type,
                    "event_date": e.event_date
                }
                for e in events
            ]
        finally:
            session.close()
    
    async def _end_current_activity(self, user_id: str):
        """结束当前活动"""
        session = get_db_session()
        try:
            state = session.query(EmotionState).filter_by(user_id=user_id).first()
            if state and state.current_activity:
                # 更新活动日志
                log = session.query(ActivityLog).filter(
                    ActivityLog.user_id == user_id,
                    ActivityLog.end_time == None
                ).order_by(ActivityLog.start_time.desc()).first()
                
                if log:
                    log.end_time = datetime.now()
                    log.duration_minutes = int((log.end_time - log.start_time).total_seconds() / 60)
                
                # 清除当前活动
                state.current_activity = None
                state.current_activity_start = None
                
                session.commit()
        finally:
            session.close()
```

```python
        context_parts.append(f"\n[当前状态: {emotion_context}]")
        
        # 添加随机提示（根据情感状态）
        emotion_state = await self.emotion_service.get_emotion_state(user_id)
        hint = self._get_emotion_based_hint(emotion_state)
        context_parts.append(f"[{hint}]")
        
        final_context = "\n".join(context_parts)
        
        # 5. 调用AI（现有逻辑）
        messages = [
            {"role": "system", "content": self._build_system_prompt()},
            {"role": "user", "content": final_context + "\n\n" + user_message}
        ]
        
        response = await self.deepseek_client.chat(messages)
        
        # 6. 更新情感状态（聊天会减少孤独感，消耗体力）
        await self.emotion_service.update_emotion_state(user_id, {
            "loneliness": -5,
            "energy": -2,
            "intimacy_level": 0.01
        })
        
        # 更新最后聊天时间
        state = await self.emotion_service.get_emotion_state(user_id)
        session = get_db_session()
        try:
            state.last_chat_time = datetime.now()
            session.add(state)
            session.commit()
        finally:
            session.close()
        
        return response
    
    def _get_emotion_based_hint(self, state: EmotionState) -> str:
        """根据情感状态生成提示"""
        if state.energy < 30:
            return "⚠️ 体力较低，回复可以提到有点累"
        elif state.sleepiness > 60:
            return "⚠️ 有点困，可以表现得迷糊一点"
        elif state.pleasure > 0.5:
            return "⚠️ 心情不错，可以稍微活泼一点"
        elif state.loneliness > 60:
            return "⚠️ 有点寂寞，可以表现出希望有人陪"
        else:
            return "⚠️ 自然回应，避免重复"
```
