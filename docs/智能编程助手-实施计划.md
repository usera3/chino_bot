```python
class ProjectManager:
    def __init__(self, project_root: str):
        self.project_root = os.path.realpath(project_root)
    
    def validate_path(self, path: str) -> bool:
        """验证路径是否在项目内"""
        real_path = os.path.realpath(os.path.join(self.project_root, path))
        return real_path.startswith(self.project_root)
    
    def read_file(self, path: str) -> str:
        """读取文件"""
        if not self.validate_path(path):
            raise SecurityError("路径不在项目内")
```

```python
# core/security.py

SAFE_COMMANDS = {
    # 查看类
    "ls", "cat", "head", "tail", "grep", "find", "tree",
    
    # Python 类
    "python", "python3", "pip list", "pip show",
    
    # 测试类
    "pytest", "python -m pytest",
    
    # Git 类（只读）
    "git status", "git log", "git diff", "git show",
}

def validate_command(command: str) -> tuple[bool, str]:
    """验证命令是否安全"""
    cmd_parts = command.split()
    base_cmd = cmd_parts[0]
    
    if base_cmd in DANGEROUS_COMMANDS:
        return False, f"危险命令：{base_cmd}"
    
    if base_cmd not in SAFE_COMMANDS:
        return False, f"命令不在白名单：{base_cmd}"
    
    return True, "OK"
```

```python
class TaskPlanner:
    def plan_task(self, description: str) -> list[dict]:
        """分解编程任务"""
        # 1. 理解任务描述
        task_type = self._classify_task(description)
        
        # 2. 生成步骤
        if task_type == "add_tool":
            return self._plan_add_tool(description)
        elif task_type == "fix_bug":
            return self._plan_fix_bug(description)
        # ...
    
    def _plan_add_tool(self, description: str) -> list[dict]:
```
